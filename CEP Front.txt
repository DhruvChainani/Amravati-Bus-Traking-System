        let progress = raw < 0.5 ? raw * 2 : 2 - raw * 2;
        progress = Math.min(Math.max(progress, 0.01), 0.99);
        const pos = getBusPosition(route, progress);
        const icon = L.divIcon({
          className: 'bus-marker',
          html: `<div style="background:${route.color}; width:18px; height:18px; border-radius:50%; border:3px solid white; box-shadow: 0 0 0 2px ${route.color}, 0 4px 12px rgba(0,0,0,0.25); display:flex; align-items:center; justify-content:center; font-size:8px; color:white; font-weight:700;">${typeof route.id === 'string' ? route.id : route.id}</div>`,
          iconSize: [22,22],
          iconAnchor: [11,11]
        });
        const marker = L.marker(pos, { icon }).addTo(busMarkerLayer);
        marker.bindPopup(`<b>${route.name}</b><br>${route.stops[0]} → ${route.stops[route.stops.length-1]}`);
      });
      if (selectedRouteId !== null) {
        const route = routesData.find(r => r.id === selectedRouteId);
        if (route) {
          const bounds = L.latLngBounds(route.stops.map(s => getCoords(s)));
          map.fitBounds(bounds, { padding: [35,35] });
        }
      } else {
        const allCoords = routesData.flatMap(r => r.stops.map(s => getCoords(s)));
        map.fitBounds(L.latLngBounds(allCoords), { padding: [30,30] });
      }
    }

    function renderSideMenu() {
      const container = document.getElementById('stopListContainer');
      const menuTitle = document.getElementById('menuRouteName');
      if (selectedRouteId === null) {
        menuTitle.textContent = 'Stops';
        container.innerHTML = `<div class="empty-menu"><i class="fas fa-hand-pointer"></i> Select a route</div>`;
        return;
      }
      const route = routesData.find(r => r.id === selectedRouteId);
      if (!route) { container.innerHTML = `<div class="empty-menu">Not found</div>`; return; }
      menuTitle.textContent = route.name;
      const period = 24;
      const raw = (simTime / period) % 1;
      let progress = raw < 0.5 ? raw * 2 : 2 - raw * 2;
      progress = Math.min(Math.max(progress, 0.01), 0.99);
      const totalStops = route.stops.length;
      const currentStopIndex = Math.min(Math.floor(progress * (totalStops - 1)), totalStops - 1);
      let html = '';
      route.stops.forEach((stop, idx) => {
        let etaMin = idx >= currentStopIndex ? Math.round((idx - currentStopIndex) * 1.8 + 0.8) : -1;
        const isCurrent = (idx === currentStopIndex);
        const etaText = etaMin < 0 ? '✓' : (etaMin === 0 ? 'now' : etaMin + 'm');
        const cls = isCurrent ? 'stop-item current' : 'stop-item';
        html += `<div class="${cls}"><span class="stop-name">${stop}</span><span class="eta">${etaText}</span></div>`;
      });
      container.innerHTML = html;
    }

    // ---------- CHIPS ----------
    function renderChips() {
      const container = document.getElementById('routeChipsContainer');
      container.innerHTML = '';
      const allChip = document.createElement('div');
      allChip.className = 'route-chip' + (selectedRouteId === null ? ' active-chip' : '');
      allChip.innerHTML = '<i class="fas fa-globe-asia"></i> All';
      allChip.dataset.routeId = 'all';
      allChip.addEventListener('click', () => { selectedRouteId = null; updateUI(); renderMap(); renderSideMenu(); });
      container.appendChild(allChip);
      routesData.forEach(route => {
        const chip = document.createElement('div');
        const isActive = (selectedRouteId === route.id);
        chip.className = 'route-chip' + (isActive ? ' active-chip' : '');
        chip.innerHTML = `<i class="fas fa-bus" style="color:${isActive ? '#f9d77e' : route.color};"></i> ${route.short}`;
        chip.dataset.routeId = route.id;
        chip.addEventListener('click', () => { selectedRouteId = route.id; updateUI(); renderMap(); renderSideMenu(); });
        container.appendChild(chip);
      });
    }

    function updateUI() {
      document.querySelectorAll('.route-chip').forEach(chip => {
        const id = chip.dataset.routeId;
        if (id === 'all' && selectedRouteId === null) chip.classList.add('active-chip');
        else if (id !== 'all' && Number(id) === selectedRouteId) chip.classList.add('active-chip');
        else chip.classList.remove('active-chip');
      });
      document.getElementById('routeNameDisplay').textContent = selectedRouteId === null ? 'All routes' : (routesData.find(r => r.id === selectedRouteId)?.name || 'Route');
    }

    // ---------- QUICK ROUTES (home) ----------
    function renderQuickRoutes() {
      const container = document.getElementById('quickRoutes');
      container.innerHTML = '';
      routesData.slice(0, 6).forEach(route => {
        const btn = document.createElement('div');
        btn.className = 'quick-route-btn';
        btn.innerHTML = `<i class="fas fa-bus" style="color:${route.color}"></i> ${route.short}`;
        btn.addEventListener('click', () => {
          selectedRouteId = route.id;
          switchPage('routes');
          setTimeout(() => { renderChips(); updateUI(); renderMap(); renderSideMenu(); }, 50);
        });
        container.appendChild(btn);
      });
    }

    // ---------- PLANNER ----------
    function populateSelects() {
      const origin = document.getElementById('originSelect');
      const dest = document.getElementById('destSelect');
      origin.innerHTML = '<option value="">Select origin...</option>';
      dest.innerHTML = '<option value="">Select destination...</option>';
      allStops.forEach(s => {
        origin.innerHTML += `<option value="${s}">${s}</option>`;
        dest.innerHTML += `<option value="${s}">${s}</option>`;
      });
    }

    document.getElementById('planBtn').addEventListener('click', function() {
      const origin = document.getElementById('originSelect').value;
      const dest = document.getElementById('destSelect').value;
      const resultDiv = document.getElementById('plannerResult');
      if (!origin || !dest) {
        resultDiv.classList.remove('show');
        alert('Please select both origin and destination.');
        return;
      }
      if (origin === dest) {
        resultDiv.classList.remove('show');
        alert('Origin and destination cannot be the same.');
        return;
      }
      // Find route that contains both stops in order
      let foundRoute = null;
      for (let route of routesData) {
        const idxOrigin = route.stops.indexOf(origin);
        const idxDest = route.stops.indexOf(dest);
        if (idxOrigin !== -1 && idxDest !== -1 && idxOrigin < idxDest) {
          foundRoute = route;
          break;
        }
      }
      if (!foundRoute) {
        // try reverse order
        for (let route of routesData) {
          const idxOrigin = route.stops.indexOf(origin);
          const idxDest = route.stops.indexOf(dest);
          if (idxOrigin !== -1 && idxDest !== -1 && idxOrigin > idxDest) {
            foundRoute = route;
            break;
          }
        }
      }
      if (foundRoute) {
        const idxO = foundRoute.stops.indexOf(origin);
        const idxD = foundRoute.stops.indexOf(dest);
        const stopsBetween = foundRoute.stops.slice(Math.min(idxO, idxD), Math.max(idxO, idxD) + 1);
        const travelTime = Math.abs(idxD - idxO) * 2 + 2;
        document.getElementById('resultRoute').textContent = `🚌 ${foundRoute.name} (${foundRoute.short})`;
        document.getElementById('resultStops').textContent = `🛑 ${stopsBetween.join(' → ')}`;
        document.getElementById('resultEta').textContent = `⏱️ Estimated travel: ~${travelTime} min`;
        resultDiv.classList.add('show');
        // Highlight the route
        selectedRouteId = foundRoute.id;
        switchPage('routes');
        setTimeout(() => { renderChips(); updateUI(); renderMap(); renderSideMenu(); }, 50);
      } else {
        document.getElementById('resultRoute').textContent = '❌ No direct route found';
        document.getElementById('resultStops').textContent = 'Try different stops';
        document.getElementById('resultEta').textContent = '';
        resultDiv.classList.add('show');
      }
    });

    // ---------- PAGE SWITCHING ----------
    window.switchPage = function(pageId) {
      document.querySelectorAll('.page').forEach(p => p.classList.remove('active-page'));
      document.querySelectorAll('.page-nav button').forEach(b => b.classList.remove('active-page'));
      document.getElementById('page-' + pageId).classList.add('active-page');
      document.querySelector(`.page-nav button[data-page="${pageId}"]`).classList.add('active-page');
      if (pageId === 'routes') {
        setTimeout(() => { map.invalidateSize(); renderMap(); renderSideMenu(); }, 100);
      }
    };

    document.querySelectorAll('.page-nav button').forEach(btn => {
      btn.addEventListener('click', function() {
        switchPage(this.dataset.page);
      });
    });

    // ---------- CLEAR ----------
    document.getElementById('clearRouteBtn').addEventListener('click', () => {
      selectedRouteId = null;
      updateUI();
      renderMap();
      renderSideMenu();
    });

    // ---------- CLOCK ----------
    function updateClock() {
      document.getElementById('clockDisplay').textContent = new Date().toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
    }

    // ---------- TICK ----------
    function tick() {
      simTime += 0.6;
      renderMap();
      renderSideMenu();
      updateClock();
    }

    // ---------- INIT ----------
    renderQuickRoutes();
    populateSelects();
    renderChips();
    renderMap();
    renderSideMenu();
    updateClock();
    setInterval(tick, 2800);

    // Fix map on routes page load
    setTimeout(() => { map.invalidateSize(); }, 300);
    console.log('🚍 Amravati Tracker with pages & planner ready');
  })();
